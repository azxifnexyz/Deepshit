local function addToggle(page, text, default, cb)
    local b = Instance.new("TextButton")
    b.Size = UDim2.new(1, 0, 0, 30)
    b.BackgroundColor3 = default and CFG.ThemeColor or Color3.fromRGB(28,28,28)
    b.Text = text .. ": " .. (default and "ON" or "OFF")
    b.TextColor3 = default and Color3.fromRGB(0,0,0) or Color3.fromRGB(255,255,255)
    b.Font = Enum.Font.GothamBold
    b.TextSize = 12
    b.Parent = page
    local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 6); c.Parent = b
    local state = default
    b.MouseButton1Click:Connect(function()
        state = not state
        b.Text = text .. ": " .. (state and "ON" or "OFF")
        b.BackgroundColor3 = state and CFG.ThemeColor or Color3.fromRGB(28,28,28)
        b.TextColor3 = state and Color3.fromRGB(0,0,0) or Color3.fromRGB(255,255,255)
        cb(state)
    end)
    return b
end

local function addSlider(page, label, minV, maxV, default, step, cb)
    local wrap = Instance.new("Frame")
    wrap.Size = UDim2.new(1, 0, 0, 44)
    wrap.BackgroundColor3 = Color3.fromRGB(22, 22, 22)
    wrap.Parent = page
    local wc = Instance.new("UICorner"); wc.CornerRadius = UDim.new(0, 6); wc.Parent = wrap

    local lab = Instance.new("TextLabel")
    lab.Size = UDim2.new(1, -70, 0, 16)
    lab.Position = UDim2.new(0, 8, 0, 4)
    lab.BackgroundTransparency = 1
    lab.Text = label
    lab.TextColor3 = Color3.fromRGB(200, 200, 200)
    lab.Font = Enum.Font.Gotham
    lab.TextSize = 11
    lab.TextXAlignment = Enum.TextXAlignment.Left
    lab.Parent = wrap

    local val = Instance.new("TextLabel")
    val.Size = UDim2.new(0, 60, 0, 16)
    val.Position = UDim2.new(1, -66, 0, 4)
    val.BackgroundTransparency = 1
    val.Text = tostring(default)
    val.TextColor3 = Color3.fromRGB(255, 255, 255)
    val.Font = Enum.Font.GothamBold
    val.TextSize = 11
    val.Parent = wrap

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -16, 0, 8)
    track.Position = UDim2.new(0, 8, 0, 26)
    track.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    track.Parent = wrap
    local tkc = Instance.new("UICorner"); tkc.CornerRadius = UDim.new(1, 0); tkc.Parent = track

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = CFG.ThemeColor
    fill.Parent = track
    local fc = Instance.new("UICorner"); fc.CornerRadius = UDim.new(1, 0); fc.Parent = fill

    local dragging = false
    local function setA(a)
        a = math.clamp(a, 0, 1)
        local v = minV + (maxV - minV) * a
        v = math.floor(v / step + 0.5) * step
        v = math.clamp(v, minV, maxV)
        fill.Size = UDim2.new((v - minV) / math.max(maxV - minV, 1), 0, 1, 0)
        val.Text = tostring(v)
        cb(v)
    end
    setA((default - minV) / math.max(maxV - minV, 1))

    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setA((i.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1))
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
            setA((i.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1))
        end
    end)
    return wrap
end

-- ─── HELPERS ──────────────────────────────────────────────
local function isEnemy(plr)
    if not plr or plr == LP then return false end
    if CFG.TeamCheck and plr.Team and LP.Team and plr.Team == LP.Team then return false end
    return true
end

local function isAlive(plr)
    local char = plr.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return false end
    if CFG.IgnoreDead and hum.Health <= 0 then return false end
    return true
end

local function getTarget()
    local best, bd = nil, CFG.AimbotFOV
    local center = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y / 2)
    for _, plr in ipairs(Players:GetPlayers()) do
        if isEnemy(plr) and isAlive(plr) then
            local head = plr.Character:FindFirstChild(CFG.HitPart)
            if head then
                local sp, on = Cam:WorldToViewportPoint(head.Position)
                if on and sp.Z > 0 then
                    local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d < bd then bd = d; best = head end
                end
            end
        end
    end
    return best
end

local flyBV, flyBG

local function stopFly()
    if flyBV then flyBV:Destroy() flyBV = nil end
    if flyBG then flyBG:Destroy() flyBG = nil end
    local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if hum then hum.PlatformStand = false end
end

local function startFly()
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    local hum = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    stopFly()
    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(9e4, 9e4, 9e4)

-- ─── RAPID FIRE ───────────────────────────────────────────
local function applyRapid(tool)
    if not CFG.RapidFire then return end
    pcall(function()
        tool:SetAttribute("RateOfFire", CFG.RapidFireRate)
        tool:SetAttribute("rateOfFire", CFG.RapidFireRate)
        tool:SetAttribute("FireRate", CFG.RapidFireRate)
        if CFG.NoReload then
            tool:SetAttribute("ReloadTime", 0)
        end
    end)
end

local function watchContainer(container)
    for _, c in ipairs(container:GetChildren()) do
        if c:IsA("Tool") then applyRapid(c) end
    end
    container.ChildAdded:Connect(function(c)
        if c:IsA("Tool") then applyRapid(c) end
    end)
end

if LP.Character then watchContainer(LP.Character) end
if LP:FindFirstChildOfClass("Backpack") then watchContainer(LP:FindFirstChildOfClass("Backpack")) end
LP.CharacterAdded:Connect(function(c)
    task.wait(0.5)
    watchContainer(c)
end)
LP.ChildAdded:Connect(function(c)
    if c:IsA("Backpack") then watchContainer(c) end
end)

-- ─── MAGIC BULLET ─────────────────────────────────────────
local function applyMagicBullet()
    if not CFG.MagicBullet then return end
    local hrp = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if isEnemy(plr) and isAlive(plr) then
            local head = plr.Character:FindFirstChild("Head")
            local vhrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if head and vhrp then
                local dist = (vhrp.Position - hrp.Position).Magnitude
                if dist <= 1000 then
                    pcall(function()
                        head:SetAttribute("HitBy", LP.UserId)
                        head:SetAttribute("Damage", 100000 * CFG.DamageMulti)
                    end)
                end
            end
        end
    end
end

-- ─── ANTI-FLING / RAGDOLL ─────────────────────────────────
local function antiFlingHook(char)
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if not hrp then return end
    hrp.ChildAdded:Connect(function(child)
        if not CFG.AntiFling then return end
        if child:IsA("BodyAngularVelocity") or child:IsA("BodyVelocity") or child:IsA("BodyForce") then
            child:Destroy()
        end
    end)
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.StateChanged:Connect(function(_, new)
            if CFG.AntiRagdoll and new == Enum.HumanoidStateType.Physics then
                hum:ChangeState(Enum.HumanoidStateType.Running)
            end
        end)
    end
end

if LP.Character then antiFlingHook(LP.Character) end
LP.CharacterAdded:Connect(antiFlingHook)
